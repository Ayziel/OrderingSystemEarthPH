const staticCacheName = 'site-static-v4';
const dynamicCacheName = 'site-dynamic-v4';
const dbName = 'offline-sync-db';
const storeName = 'syncQueue';

const dashboardAssets = [
    "/EarthPhFrontEndWeb/Pages/System/fallback.html",
    "/EarthPhFrontEndWeb/Pages/System/index.html",
    "/EarthPhFrontEndWeb/Pages/System/scripts/dashboard.js",

    "/EarthPhFrontEndWeb/Pages/System/styles/gcash.css",
    "/EarthPhFrontEndWeb/Pages/System/styles/agentPerformance.css",
    "/images/logo.jpg",
];

const generalAssets = [
    "/",
    "/EarthPhFrontEndWeb/Pages/System/styles/style.css",
    "/EarthPhFrontEndWeb/Pages/System/styles/navbar.css",
    "/EarthPhFrontEndWeb/Pages/System/scripts/script.js",
    "/EarthPhFrontEndWeb/Pages/System/scripts/manifestShared.js",
    "/EarthPhFrontEndWeb/Pages/System/scripts/sideNavigation.js",
];

const assets = [...dashboardAssets, ...generalAssets];

// Open the IndexedDB database
const openDb = () => {
    console.log('Service worker loading...');
    return new Promise((resolve, reject) => {
        const request = indexedDB.open(dbName, 1);

        request.onupgradeneeded = (e) => {
            const db = e.target.result;
            console.log("Creating or upgrading IndexedDB");
            if (!db.objectStoreNames.contains(storeName)) {
                db.createObjectStore(storeName, { keyPath: 'id', autoIncrement: true });
                console.log('Created object store:', storeName);
            }
        };

        request.onsuccess = (e) => {
            console.log('IndexedDB opened successfully.');
            resolve(e.target.result);
        };

        request.onerror = (e) => {
            console.error('Error opening IndexedDB:', e.target.error);
            reject(e.target.error);
        };
    });
};


// Add an item to the sync queue
const addToSyncQueue = async (data) => {
    const db = await openDb();
    const tx = db.transaction(storeName, 'readwrite');
    const store = tx.objectStore(storeName);
    store.add(data);
    tx.complete.then(() => {
        console.log("Data added to IndexedDB successfully.");
    }).catch((err) => {
        console.error("Failed to add data to IndexedDB:", err);
    });
    return tx.complete;
};



// Get all items from the sync queue
const getSyncQueue = async () => {
    const db = await openDb();
    const tx = db.transaction(storeName, 'readonly');
    const store = tx.objectStore(storeName);
    const items = await store.getAll();
    return items;
};

// Clear the sync queue (once data is synced successfully)
const clearSyncQueue = async () => {
    const db = await openDb();
    const tx = db.transaction(storeName, 'readwrite');
    const store = tx.objectStore(storeName);
    store.clear();
    return tx.complete;
};

// Install event
self.addEventListener('install', (event) => {
    event.waitUntil(
        caches.open(staticCacheName).then((cache) => {
            console.log('Caching static assets...');
            return cache.addAll(assets); // Cache the predefined assets
        })
    );
});

// Activate event
self.addEventListener('activate', (event) => {
    const cacheWhitelist = [staticCacheName, dynamicCacheName];
    
    event.waitUntil(
        caches.keys().then((cacheNames) => {
            return Promise.all(
                cacheNames.map((cacheName) => {
                    if (!cacheWhitelist.includes(cacheName)) {
                        console.log('Deleting old cache', cacheName);
                        return caches.delete(cacheName);
                    }
                })
            );
        })
    );
});

// Fetch event
self.addEventListener('fetch', (event) => {
    const { request } = event;

    if (request.method === 'GET') {
        // Cache GET requests
        event.respondWith(
            caches.match(request).then((cachedResponse) => {
                if (cachedResponse) {
                    return cachedResponse; // Return cached response if available
                }

                // If not in cache, fetch from network and cache it
                return fetch(request).then((response) => {
                    return caches.open(staticCacheName).then((cache) => {
                        cache.put(request, response.clone()); // Cache the response for future use
                        return response;
                    });
                });
            })
        );
    } else if (request.method === 'POST') {
        // Handle POST requests differently, for example by storing data in IndexedDB
        // Note: POST requests should not be cached in Cache API.
        event.respondWith(
            (async () => {
                const response = await fetch(request);
                if (!response.ok) {
                    // Handle failed POST request or save it for syncing later.
                    console.error('Failed to submit POST request');
                    await addToSyncQueue({ url: request.url, method: request.method, body: await request.clone().text() }); // Save failed request
                }
                return response;
            })()
        );
    }
});

// Sync event for Background Sync
self.addEventListener('sync', (event) => {
    if (event.tag === 'sync-surveys') {
        event.waitUntil(syncData());
    }
});

// Sync data from IndexedDB
const syncData = async () => {
    const syncQueue = await getSyncQueue(); // Get unsynced data from IndexedDB

    if (syncQueue.length > 0) {
        try {
            for (let item of syncQueue) {
                // Sync the data with your server
                const response = await fetch('https://earthph.sdevtech.com.ph/survey/createSurvey', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(item),
                });

                if (response.ok) {
                    console.log('Survey synced successfully:', item);
                    await removeFromSyncQueue(item.id);  // Remove successfully synced request from queue
                } else {
                    console.error('Sync failed for:', item);
                }
            }
        } catch (error) {
            console.error('Sync error:', error);
        }
    }
};

// Remove a specific item from the sync queue after it has been synced
const removeFromSyncQueue = async (id) => {
    const db = await openDb();
    const tx = db.transaction(storeName, 'readwrite');
    const store = tx.objectStore(storeName);
    store.delete(id); // Remove the item from the queue
    return tx.complete;
};

// Example function to handle form submission
const handleFormSubmit = async (formData) => {
    console.log("Form submit called with data:", formData);  // Check if this gets logged
    if (navigator.onLine) {
        // If online, sync immediately
        await syncData();  // Sync all data to the server immediately
    } else {
        // If offline, store the data in IndexedDB for later sync
        await addToSyncQueue(formData);
        console.log('Data saved offline, will sync later');
    }
};
